# tests/kernel/atomics/expected-bytes.txt — R18-M3-003 / #770
#
# Byte-level expected encodings for the atomic primitives that
# src/kernel/core/sync/atomic.pdx emits. Consumed by
# tools/verify-atomics.sh, which disassembles build/core/sync/atomic.o
# with objdump -d and asserts each symbol's body contains the exact
# hex sequence below.
#
# The point of the witness is NOT to re-verify paideia-as's encoder tests
# (those live in tools/paideia-as/crates/paideia-as-encoder/tests/) —
# it is to catch a regression in either (a) the paideia-as build that
# gets pinned into this repo, or (b) the surrounding compiled context
# (register-allocator picking a different scratch register, prologue
# insertion, etc.) that would silently perturb the emitted bytes.
#
# Format (one non-comment line per assertion):
#   SYMBOL|EXPECTED_HEX|DESCRIPTION
#
# EXPECTED_HEX is the exact byte sequence (lowercase, space-separated)
# that must appear inside `objdump -d --disassemble=SYMBOL` output when
# the raw bytes column is grepped. objdump prints one byte per position
# separated by a single space, so the grep matches the bytes as printed.
#
# Sources of truth (Intel SDM 64/IA-32 Software Developer's Manual):
#   MOV r64, r/m64          Vol 2A §MOV                    (opcode 8B /r)
#   MOV r/m64, r64          Vol 2A §MOV                    (opcode 89 /r)
#   XCHG r/m64, r64         Vol 2A §XCHG (implicit LOCK)   (opcode 87 /r)
#   LOCK CMPXCHG r/m64, r64 Vol 2A §CMPXCHG                (F0 0F B1 /r)
#   LOCK XADD r/m64, r64    Vol 2B §XADD                   (F0 0F C1 /r)
#   MFENCE                  Vol 2B §MFENCE                 (0F AE F0)
#   RET                     Vol 2B §RET (near return)      (C3)

# atomic_load_u64: mov rax, [rdi] ; ret
#   48 8b 07 = REX.W MOV r64, r/m64 with ModR/M reg=rax, mem=[rdi]
#   c3       = ret
atomic_load_u64|48 8b 07|REX.W mov rax, [rdi] (aligned 64-bit load)

# atomic_store_u64: mov [rdi], rsi ; ret
#   48 89 37 = REX.W MOV r/m64, r64 with ModR/M reg=rsi, mem=[rdi]
#   c3       = ret
atomic_store_u64|48 89 37|REX.W mov [rdi], rsi (TSO-ordered 64-bit store)

# atomic_cas_u64: mov rax, rsi ; mov rcx, rdx ; lock cmpxchg [rdi], rcx ; ret
#   48 89 f0    = REX.W mov rax, rsi  (place expected in rax for lock_cmpxchg)
#   48 89 d1    = REX.W mov rcx, rdx  (place new in rcx as register operand)
#   f0 48 0f b1 = LOCK REX.W 0F B1    (lock cmpxchg core opcode)
#   0f          = ModR/M mem=[rdi], reg=rcx (mod=00, reg=001, r/m=111)
#   c3          = ret
atomic_cas_u64|f0 48 0f b1 0f|LOCK REX.W CMPXCHG [rdi], rcx (full-fence CAS)

# atomic_xadd_u64: mov rax, rsi ; lock xadd [rdi], rax ; ret
#   48 89 f0    = REX.W mov rax, rsi  (rax carries delta AND receives prior)
#   f0 48 0f c1 = LOCK REX.W 0F C1    (lock xadd core opcode)
#   07          = ModR/M mem=[rdi], reg=rax (mod=00, reg=000, r/m=111)
#   c3          = ret
atomic_xadd_u64|f0 48 0f c1 07|LOCK REX.W XADD [rdi], rax (full-fence fetch-and-add)

# atomic_xchg_u64: mov rax, rsi ; xchg [rdi], rax ; ret
#   48 89 f0 = REX.W mov rax, rsi     (rax carries val AND receives prior)
#   48 87 07 = REX.W 87 with ModR/M mem=[rdi], reg=rax
#              NO explicit F0 prefix: XCHG on a memory operand is implicitly
#              locked by the CPU (SDM Vol 2A §XCHG) — encoder deliberately
#              omits the F0 byte because the bus lock is always taken.
#   c3       = ret
atomic_xchg_u64|48 87 07|REX.W XCHG [rdi], rax (implicit-LOCK atomic swap)

# mem_barrier: mfence ; ret
#   0f ae f0 = MFENCE (three bytes, zero operands, no REX)
#   c3       = ret
mem_barrier|0f ae f0|MFENCE (serializing memory barrier)
